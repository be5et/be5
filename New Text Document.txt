  // باقي الكود بدون تغيير ...
  String currentWord = "";
  String question = "";
  String hint = "";
  int currentLevel = 0;
  int score = 0;
  List<String> guessedLetters = [];
  List<String> revealedLetters = [];
  bool showHint = false;

  late BannerAd bannerAd;
  late InterstitialAd interstitialAd;
  late RewardedAd rewardedAd;
  late RewardedInterstitialAd rewardedInterstitialAd;
  bool isBannerAdLoaded = false;
  bool isInterstitialAdLoaded = false;
  bool isRewardedAdLoaded = false;
  bool isRewardedInterstitialAdLoaded = false;

  @override
  void initState() {
    super.initState();
    loadAds();
    loadLevel();
    showAppOpenAd();
  }

  void loadAds() {
    bannerAd = BannerAd(
      adUnitId: "ca-app-pub-9244590788989680/3701917807",
      size: AdSize.banner,
      request: AdRequest(),
      listener: BannerAdListener(
        onAdLoaded: (Ad ad) {
          setState(() {
            isBannerAdLoaded = true;
          });
        },
        onAdFailedToLoad: (Ad ad, LoadAdError error) {
          ad.dispose();
        },
      ),
    )..load();

    InterstitialAd.load(
      adUnitId: "ca-app-pub-9244590788989680/9629031908",
      request: AdRequest(),
      adLoadCallback: InterstitialAdLoadCallback(
        onAdLoaded: (InterstitialAd ad) {
          interstitialAd = ad;
          isInterstitialAdLoaded = true;
        },
        onAdFailedToLoad: (LoadAdError error) {
          isInterstitialAdLoaded = false;
        },
      ),
    );

    RewardedAd.load(
      adUnitId: "ca-app-pub-9244590788989680/3063623554",
      request: AdRequest(),
      rewardedAdLoadCallback: RewardedAdLoadCallback(
        onAdLoaded: (RewardedAd ad) {
          rewardedAd = ad;
          isRewardedAdLoaded = true;
        },
        onAdFailedToLoad: (LoadAdError error) {
          isRewardedAdLoaded = false;
        },
      ),
    );

    RewardedInterstitialAd.load(
      adUnitId: "ca-app-pub-9244590788989680/7002868562",
      request: AdRequest(),
      rewardedInterstitialAdLoadCallback: RewardedInterstitialAdLoadCallback(
        onAdLoaded: (RewardedInterstitialAd ad) {
          rewardedInterstitialAd = ad;
          isRewardedInterstitialAdLoaded = true;
        },
        onAdFailedToLoad: (LoadAdError error) {
          isRewardedInterstitialAdLoaded = false;
        },
      ),
    );
  }

  void showAppOpenAd() {
    AppOpenAd.load(
      adUnitId: "ca-app-pub-9244590788989680/8578839963",
      request: AdRequest(),
      adLoadCallback: AppOpenAdLoadCallback(
        onAdLoaded: (AppOpenAd ad) {
          ad.show();
        },
        onAdFailedToLoad: (LoadAdError error) {
          // فشل تحميل الإعلان
        },
      ),
      orientation: AppOpenAd.orientationPortrait,
    );
  }

  void loadLevel() {
    setState(() {
      currentWord = pyramidWords[currentLevel]["word"]!;
      question = pyramidWords[currentLevel]["question"]!;
      hint = pyramidWords[currentLevel]["hint"]!;
      guessedLetters.clear();
      revealedLetters.clear();
      showHint = false;
    });
  }

  void handleLetterGuess(String letter) {
    if (!guessedLetters.contains(letter)) {
      setState(() {
        guessedLetters.add(letter);
        if (currentWord.contains(letter)) {
          score += 10;
        }
      });
    }
  }

  void skipLevel() {
    if (currentLevel < pyramidWords.length - 1) {
      if (isInterstitialAdLoaded) {
        interstitialAd.show();
      }
      setState(() {
        currentLevel++;
        score = (score >= 20) ? score - 20 : 0;
      });
      loadLevel();
    }
  }

  void revealHint() {
    if (isRewardedAdLoaded) {
      rewardedAd.show(
        onUserEarnedReward: (AdWithoutView ad, RewardItem reward) {
          setState(() {
            showHint = true;
          });
        },
      );
    }
  }

  void revealLetter() {
    if (isRewardedInterstitialAdLoaded) {
      rewardedInterstitialAd.show(
        onUserEarnedReward: (AdWithoutView ad, RewardItem reward) {
          final unrevealedLetters = currentWord
              .split('')
              .where((letter) => !guessedLetters.contains(letter) && !revealedLetters.contains(letter))
              .toList();
          if (unrevealedLetters.isNotEmpty) {
            setState(() {
              revealedLetters.add(unrevealedLetters.first);
            });
          }
        },
      );
    }
  }

  @override
  void dispose() {
    bannerAd.dispose();
    super.dispose();
  }
